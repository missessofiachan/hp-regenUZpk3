version "4.11"

class HealthRegenHandler : EventHandler
{
    int damageCooldown[MAXPLAYERS];

    CVar enabledCVar;
    CVar capCVar;
    CVar amountCVar;
    CVar modeCVar;
    CVar segmentSizeCVar;
    CVar onlyStillCVar;
    CVar crouchMultCVar;
    CVar stillMultCVar;
    CVar delayCVar;
    CVar particlesCVar;
    CVar soundCVar;

    override void OnRegister()
    {
        enabledCVar = CVar.FindCVar("mod_regen_enabled");
        capCVar = CVar.FindCVar("mod_regen_cap");
        amountCVar = CVar.FindCVar("mod_regen_amount");
        modeCVar = CVar.FindCVar("mod_regen_mode");
        segmentSizeCVar = CVar.FindCVar("mod_regen_segment_size");
        onlyStillCVar = CVar.FindCVar("mod_regen_only_still");
        crouchMultCVar = CVar.FindCVar("mod_regen_crouch_mult");
        stillMultCVar = CVar.FindCVar("mod_regen_still_mult");
        delayCVar = CVar.FindCVar("mod_regen_delay");
        particlesCVar = CVar.FindCVar("mod_regen_particles");
        soundCVar = CVar.FindCVar("mod_regen_sound");
    }

    override void WorldTick()
    {
        // Decrement damage cooldowns every single tick (35 tics = 1 second)
        for (int i = 0; i < MAXPLAYERS; i++)
        {
            if (damageCooldown[i] > 0)
            {
                damageCooldown[i]--;
            }
        }

        // Halt operations immediately if the module is disabled
        if (enabledCVar && !enabledCVar.GetBool())
        {
            return;
        }

        // Execute core health evaluations exactly once per second
        if (level.time % 35 != 0)
        {
            return;
        }

        double capPercentage = (capCVar ? capCVar.GetInt() : 25) / 100.0;
        int baseHealAmount = amountCVar ? amountCVar.GetInt() : 5;
        int regenMode = modeCVar ? modeCVar.GetInt() : 0;

        for (int i = 0; i < MAXPLAYERS; i++)
        {
            if (playeringame[i] && players[i].mo)
            {
                PlayerPawn player = players[i].mo;
                int maxHealth = player.GetMaxHealth();
                int regenCap = maxHealth;

                // --- 1. HEALTH CEILING EVALUATION ---
                if (regenMode == 0) // Standard Cap %
                {
                    regenCap = int(maxHealth * capPercentage);
                }
                else if (regenMode == 1) // Segment-Based Mode
                {
                    int currentHealth = player.health;
                    int segmentSize = segmentSizeCVar ? max(1, segmentSizeCVar.GetInt()) : 20; 
                    
                    int nextSegment = ((currentHealth + segmentSize - 1) / segmentSize) * segmentSize;
                    regenCap = nextSegment < maxHealth ? nextSegment : maxHealth;
                }

                // Honor incoming active damage cooldowns
                if (damageCooldown[i] > 0)
                {
                    continue;
                }

                // --- 2. MOVEMENT RESTRICTION CHECK ---
                bool isStill = player.vel.xy.Length() < 0.1;
                if (onlyStillCVar && onlyStillCVar.GetBool() && !isStill)
                {
                    continue;
                }

                int finalHealAmount = baseHealAmount;

                // --- 3. POSTURE AND STILLNESS MULTIPLIERS ---
                double playerStateMultiplier = 1.0;
                if (players[i].crouchfactor < 1.0 || players[i].crouchdir != 0)
                {
                    if (crouchMultCVar) playerStateMultiplier *= crouchMultCVar.GetFloat();
                }

                if (isStill)
                {
                    if (stillMultCVar) playerStateMultiplier *= stillMultCVar.GetFloat();
                }

                finalHealAmount = int(finalHealAmount * playerStateMultiplier);

                // --- 4. EXECUTE BODY RESTORATION ---
                if (player.health > 0 && player.health < regenCap)
                {
                    int oldHealth = player.health;
                    player.GiveBody(max(1, finalHealAmount), regenCap);
                    
                    if (player.health > oldHealth)
                    {
                        // Audio response
                        if (soundCVar && soundCVar.GetBool())
                        {
                            player.A_StartSound("misc/i_pk_up", CHAN_ITEM, CHANF_LOCAL, 0.3);
                        }
                        
                        // Particle FX response
                        if (particlesCVar && particlesCVar.GetBool())
                        {
                            for (int j = 0; j < 12; j++)
                            {
                                double angle = FRandom[Regen](0, 360);
                                double dist = FRandom[Regen](10, 20);
                                double vx = Cos(angle) * FRandom[Regen](0.1, 0.4);
                                double vy = Sin(angle) * FRandom[Regen](0.1, 0.4);
                                double vz = FRandom[Regen](0.2, 0.8);
                                player.A_SpawnParticle(
                                    "00FF00",
                                    SPF_FULLBRIGHT,
                                    35,
                                    FRandom[Regen](2, 3),
                                    angle,
                                    dist, 0, FRandom[Regen](8, player.height - 8),
                                    vx, vy, vz
                                );
                            }
                        }
                    }
                }
            }
        }
    }

    override void WorldThingDamaged(WorldEvent e)
    {
        if (e.Thing && e.Thing.player)
        {
            int playernum = e.Thing.PlayerNumber();
            if (playernum >= 0 && playernum < MAXPLAYERS)
            {
                int customDelay = delayCVar ? delayCVar.GetInt() : 5;
                
                // Log down the total ticks remaining (Seconds * 35 Engine Tics)
                damageCooldown[playernum] = customDelay * 35;
            }
        }
    }
}
